cmake_minimum_required(VERSION 3.22)
project(Kalipso VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Network DBus)
add_executable(kalipso-engine src/engine/main.cpp src/speech.cpp src/moonshine.cpp)
target_include_directories(kalipso-engine PRIVATE src)
target_link_libraries(kalipso-engine PRIVATE Qt6::Core Qt6::DBus)
target_compile_options(kalipso-engine PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS kalipso-engine RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT Engine)
configure_file(packaging/io.github.brianidavide.Kalipso.Engine.service.in
  io.github.brianidavide.Kalipso.Engine.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/io.github.brianidavide.Kalipso.Engine.service
  DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/services COMPONENT Engine)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/kalipso-engine COMPONENT Engine)
add_library(prompter_core src/tracker.cpp src/approval.cpp)
target_include_directories(prompter_core PUBLIC src)
target_link_libraries(prompter_core PUBLIC Qt6::Core)
target_compile_options(prompter_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(kalipso src/main.cpp src/window.cpp src/speech.cpp src/moonshine.cpp src/llm.cpp)
target_link_libraries(kalipso PRIVATE prompter_core Qt6::Widgets Qt6::Network)
target_compile_options(kalipso PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS kalipso RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.github.brianidavide.kalipso.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/io.github.brianidavide.kalipso.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
install(FILES packaging/io.github.brianidavide.kalipso.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/kalipso)
if(BUILD_TESTING)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME speech_assets COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/speech_assets_test.py)
  endif()
  find_program(DBUS_RUN_SESSION dbus-run-session)
  find_program(GDBUS_EXECUTABLE gdbus)
  if(Python3_Interpreter_FOUND AND DBUS_RUN_SESSION AND GDBUS_EXECUTABLE)
    add_test(NAME engine_dependencies COMMAND ${DBUS_RUN_SESSION} -- ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/engine_dependencies.py $<TARGET_FILE:kalipso-engine>)
  endif()
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE prompter_core Qt6::Test)
  add_test(NAME core COMMAND core_tests)
  add_executable(network_tests tests/network_tests.cpp src/llm.cpp)
  target_link_libraries(network_tests PRIVATE prompter_core Qt6::Network Qt6::Test)
  add_test(NAME network COMMAND network_tests)
  add_executable(ui_tests tests/ui_tests.cpp src/window.cpp src/speech.cpp src/moonshine.cpp src/llm.cpp)
  target_link_libraries(ui_tests PRIVATE prompter_core Qt6::Widgets Qt6::Network Qt6::Test)
  add_test(NAME ui COMMAND ui_tests)
  set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_library(fake_moonshine SHARED tests/fake_moonshine.cpp)
  add_executable(speech_tests tests/speech_tests.cpp src/speech.cpp src/moonshine.cpp)
  target_link_libraries(speech_tests PRIVATE Qt6::Core Qt6::Test)
  target_include_directories(speech_tests PRIVATE src)
  target_compile_definitions(speech_tests PRIVATE FAKE_MOONSHINE_LIBRARY="$<TARGET_FILE:fake_moonshine>")
  add_dependencies(speech_tests fake_moonshine)
  add_test(NAME speech COMMAND speech_tests)
  add_executable(speech_probe benchmarks/speech_probe.cpp src/speech.cpp src/moonshine.cpp)
  target_link_libraries(speech_probe PRIVATE Qt6::Core)
  target_include_directories(speech_probe PRIVATE src)
endif()
option(BUILD_RENDER_BENCHMARKS "Build equivalent Widgets/Quick render probes" OFF)
if(BUILD_RENDER_BENCHMARKS)
  find_package(Qt6 REQUIRED COMPONENTS Quick)
  add_executable(render_probe benchmarks/render_probe.cpp)
  target_link_libraries(render_probe PRIVATE Qt6::Widgets Qt6::Quick)
endif()
