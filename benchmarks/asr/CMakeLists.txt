cmake_minimum_required(VERSION 3.22)
project(KalipsoAsrBench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core)
add_executable(vosk_bench native_worker.cpp)
target_link_libraries(vosk_bench PRIVATE Qt6::Core)
if(WHISPER_ROOT)
  add_executable(whisper_bench native_worker.cpp)
  target_compile_definitions(whisper_bench PRIVATE BENCH_WHISPER)
  target_include_directories(whisper_bench PRIVATE ${WHISPER_ROOT}/include ${WHISPER_ROOT}/ggml/include)
  target_link_directories(whisper_bench PRIVATE ${WHISPER_ROOT}/build/bin)
  target_link_libraries(whisper_bench PRIVATE Qt6::Core whisper)
  set_target_properties(whisper_bench PROPERTIES BUILD_RPATH "${WHISPER_ROOT}/build/bin")
endif()
